- DeepSeek成本极低（¥0.001/1K tokens）
- 100活跃用户月成本仅¥40左右

**智能路由**：
- 80%日常任务 → DeepSeek（成本极低）
- 15%备份场景 → 通义千问（免费额度/主力故障）

**无需运维**：
- 第三方服务，自动更新
- 无需GPU管理
- 自动扩展，无限并发

### 2.2 统一API适配器

```typescript
interface AIServiceAdapter {
  name: string;
  generateCode(prompt: string, options: GenerateOptions): AsyncIterableIterator<string>;
  explainCode(code: string): Promise<string>;
  optimizeCode(code: string): Promise<string>;
}

class DeepSeekAdapter implements AIServiceAdapter {
  name = 'DeepSeek';
  
  async *generateCode(prompt: string, options: GenerateOptions) {
    const response = await this.client.chat.completions.create({
      model: 'deepseek-coder',
      messages: [{ role: 'user', content: prompt }],
      stream: true,
      max_tokens: options.maxTokens,
    });
    
    for await (const chunk of response) {
      yield chunk.choices[0]?.delta?.content || '';
    }
  }
}
```

### 2.3 智能路由策略

```typescript
class AIRouter {
  selectService(task: AITask, user: User): AIServiceAdapter {
    // 1. 付费用户的复杂任务 → GPT-4
    if (user.tier === 'paid' && task.complexity === 'high') {
      return this.gpt4Adapter;
    }
    
    // 2. 检查DeepSeek可用性（主力）
    if (await this.deepseekAdapter.isHealthy()) {
      return this.deepseekAdapter;
    }
    
    // 3. 降级到通义千问（备用）
    if (await this.qwenAdapter.isHealthy()) {
      return this.qwenAdapter;
    }
    
    // 4. 最后降级到GPT-3.5
    return this.gpt35Adapter;
  }
  
  // 任务复杂度评估
  evaluateComplexity(userInput: string): 'low' | 'medium' | 'high' {
    const length = userInput.length;
    const keywords = ['循环', '异步', '错误处理', '多线程'];
    
    if (length > 500 || keywords.some(k => userInput.includes(k))) {
      return 'high';
    }
    return length > 200 ? 'medium' : 'low';
  }
}
```

### 3.2 上下文动态组装

```typescript
class PromptBuilder {
  build(template: PromptTemplate, context: UserContext): string {
    let prompt = template.systemPrompt;
    
    // 1. 注入设备上下文
    if (context.device) {
      prompt += `\n当前设备: ${context.device.type}`;
      prompt += `\n协议: ${context.device.protocol}`;
    }
    
    // 2. 注入串口配置
    prompt += `\n波特率: ${context.serialConfig.baudRate}`;
    
    // 3. 注入历史命令（最近10条）
    if (context.recentCommands.length > 0) {
      prompt += `\n最近命令: ${context.recentCommands.join(', ')}`;
    }
    
    // 4. 注入Few-Shot示例
    prompt += '\n\n示例：\n';
    for (const example of template.fewShotExamples) {
      prompt += `输入: ${example.input}\n`;
      prompt += `输出: ${example.output}\n\n`;
    }
    
    // 5. 注入用户请求
    prompt += `\n现在请根据以下需求生成代码：\n${context.userInput}`;
    
    return prompt;
  }
}
```

## 4. 流式响应架构

### 4.1 后端实现（Server-Sent Events）

```typescript
// Express路由
app.post('/api/ai/generate', async (req, res) => {
  // 设置SSE响应头
  res.setHeader('Content-Type', 'text/event-stream');
  res.setHeader('Cache-Control', 'no-cache');
  res.setHeader('Connection', 'keep-alive');
  
  try {
    // 智能路由选择服务
    const service = aiRouter.selectService(req.body.task, req.user);
    
    // 获取AI流式响应（第三方API）
    const stream = await service.generateCode(
      req.body.prompt,
      { maxTokens: req.body.maxTokens }
    );
    
    // 流式发送Token
    for await (const token of stream) {
      res.write(`data: ${JSON.stringify({ token })}\n\n`);
    }
    
    // 发送完成事件
    res.write(`data: ${JSON.stringify({ done: true })}\n\n`);
    res.end();
  } catch (error) {
    res.write(`data: ${JSON.stringify({ error: error.message })}\n\n`);
    res.end();
  }
});
```

### 4.2 前端实现（EventSource）

```typescript
function useAIStream(prompt: string, context: Context) {
  const [response, setResponse] = useState('');
  const [isLoading, setIsLoading] = useState(false);
  
  const generate = useCallback(() => {
    setIsLoading(true);
    setResponse('');
    
    // 创建SSE连接
    const eventSource = new EventSource('/api/ai/generate', {
      method: 'POST',
      body: JSON.stringify({ prompt, context }),
    });
    
    // 监听消息
    eventSource.onmessage = (event) => {
      const data = JSON.parse(event.data);
      
      if (data.done) {
        eventSource.close();
        setIsLoading(false);
      } else if (data.token) {
        setResponse((prev) => prev + data.token);
      }
    };
    
    // 错误处理
    eventSource.onerror = () => {
      eventSource.close();
      setIsLoading(false);
    };
  }, [prompt, context]);
  
  return { response, isLoading, generate };
}
```

```typescript
class DeviceRecognitionService {
  // 通过AT命令响应识别设备
  recognizeByATResponse(response: string): DeviceInfo | null {
    if (response.includes('ESP32')) {
      return {
        type: 'ESP32',
        protocol: 'AT_COMMANDS',
        knownCommands: ESP32_AT_COMMANDS,
      };
    }
    if (response.includes('ESP8266')) {
      return {

        protocol: 'AT_COMMANDS',
```

### 6.2 限额检查与队列管理

```typescript
class RateLimitService {
  async checkAndConsume(userId: string): Promise<boolean> {
    const tier = await this.getUserTier(userId);
    const limits = RATE_LIMIT_TIERS[tier];
    
    // 检查每日限额
    const dailyUsage = await redis.get(`ai:daily:${userId}`);
    if (dailyUsage >= limits.dailyLimit) {
      throw new Error('已达到每日限额');
    }
    
    // 检查每分钟限额
    const minuteUsage = await redis.get(`ai:minute:${userId}`);
    if (minuteUsage >= limits.minuteLimit) {
      throw new Error('请求过于频繁，请稍后再试');
    }
    
    // 消费配额
    await redis.incr(`ai:daily:${userId}`);
    await redis.incr(`ai:minute:${userId}`);
    await redis.expire(`ai:minute:${userId}`, 60);
    
    return true;
  }
  
  // 优先级队列
  async enqueue(userId: string, request: AIRequest): Promise<void> {
    const tier = await this.getUserTier(userId);
    const priority = RATE_LIMIT_TIERS[tier].priority;
    
    // 根据优先级加入不同队列
    await redis.zadd(`ai:queue:${priority}`, Date.now(), 
                     JSON.stringify(request));
  }
}
```

## 7. 成本优化策略

### 7.1 智能缓存

```typescript
class AIResponseCache {
  async getCached(prompt: string, context: Context): Promise<string | null> {
    // 1. 标准化Prompt（移除个性化信息）
    const normalizedPrompt = this.normalize(prompt, context);
    
    // 2. 计算哈希
    const hash = this.hash(normalizedPrompt);
    
    // 3. 查询缓存
    const cached = await redis.get(`ai:cache:${hash}`);
    if (cached) {
      // 4. 个性化替换
      return this.personalize(cached, context);
    }
    
    return null;
  }
  
  async setCached(prompt: string, context: Context, response: string) {
    const normalizedPrompt = this.normalize(prompt, context);
    const hash = this.hash(normalizedPrompt);
    
    // 缓存7天
    await redis.setex(`ai:cache:${hash}`, 7 * 24 * 3600, response);
  }
}
```

**缓存命中率优化**：
