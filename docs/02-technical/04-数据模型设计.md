## 2. 完整 Prisma Schema

### 2.1 Schema 文件位置

文件路径：`backend/prisma/schema.prisma`

### 2.2 完整 Schema 定义

```prisma
// This is your Prisma schema file,
// learn more about it in the docs: https://pris.ly/d/prisma-schema

generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

// ============================================
// 用户模型
// ============================================
model User {
  id              String   @id @default(uuid())
  email           String   @unique @db.VarChar(255)
  passwordHash    String?  @db.VarChar(255)
  nickname        String?  @db.VarChar(100)
  avatar          String?  @db.VarChar(500)
  authProvider    String   @default("email") @db.VarChar(20) // 'email', 'google', 'github'
  isEmailVerified Boolean  @default(false)
  lastLoginAt     DateTime?
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt

  // 关联关系
  profiles     SerialProfile[]
  macros       Macro[]
  histories    SendHistory[]
  preference   UserPreference?
  sessions     SessionParticipant[]
  ownedSessions Session[] @relation("SessionOwner")

  // 索引
  @@index([email])
  @@index([authProvider])
  @@index([createdAt])
  @@map("users")
}

// ============================================
// 串口配置模型
// ============================================
model SerialProfile {
  id              String   @id @default(uuid())
  userId          String   @db.Uuid
  name            String   @db.VarChar(100) // 'default' 或设备别名
  deviceId        String?  @db.VarChar(50) // USB Vendor:Product ID (如: "1A86:7523")
  baudRate        Int      @default(115200)
  dataBits        Int      @default(8) // 5, 6, 7, 8
  parity          String   @default("none") @db.VarChar(10) // 'none', 'even', 'odd', 'mark', 'space'
  stopBits        Float    @default(1.0) // 1, 1.5, 2
  flowControl     String   @default("none") @db.VarChar(10) // 'none', 'hardware', 'software'
  lineEndingSend  String   @default("none") @db.VarChar(10) // 'none', 'cr', 'lf', 'crlf'
  lineEndingParse String   @default("auto") @db.VarChar(10) // 'none', 'cr', 'lf', 'crlf', 'auto'
  encoding        String   @default("utf-8") @db.VarChar(20) // 'utf-8', 'ascii', 'hex'
  bufferSize      Int      @default(65536)
  isDefault       Boolean  @default(false)
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt

  // 关联关系
  user User @relation(fields: [userId], references: [id], onDelete: Cascade)

  // 索引
  @@index([userId])
  @@index([userId, isDefault])
  @@index([deviceId])
  @@unique([userId, name])
  @@map("serial_profiles")
}

// ============================================
// 宏命令模型
// ============================================
model Macro {
  id              String   @id @default(uuid())
  userId          String   @db.Uuid
  name            String   @db.VarChar(100)
  category        String   @default("default") @db.VarChar(50)
  contentTemplate String   @db.Text // 支持 {variable} 插值
  variablesSchema Json?    // 变量定义数组
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt

  // 关联关系
  user User @relation(fields: [userId], references: [id], onDelete: Cascade)

  // 索引
  @@index([userId])
  @@index([userId, category])
  @@map("macros")
}

// ============================================
// 发送历史模型
// ============================================
model SendHistory {
  id        String   @id @default(uuid())
  userId    String   @db.Uuid
  content   String   @db.Text
  encoding  String   @default("utf-8") @db.VarChar(20)
  createdAt DateTime @default(now())

  // 关联关系
  user User @relation(fields: [userId], references: [id], onDelete: Cascade)

  // 索引
  @@index([userId])
  @@index([userId, createdAt])
  @@map("send_histories")
}

// ============================================
// 用户偏好模型
// ============================================
model UserPreference {
  userId          String   @id @db.Uuid
  theme           String   @default("auto") @db.VarChar(10) // 'light', 'dark', 'auto'
  fontSize        Int      @default(14) // 12, 14, 16, 18
  fontFamily      String   @default("Monaco") @db.VarChar(100)
  lineHeight      String   @default("normal") @db.VarChar(20) // 'compact', 'normal', 'comfort'
  autoScroll      Boolean  @default(true)
  showLineNumber  Boolean  @default(true)
  timestampFormat String   @default("HH:mm:ss.SSS") @db.VarChar(20)
  highlightRules  Json?    // 关键字高亮规则数组
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt

  // 关联关系
  user User @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@map("user_preferences")
}

// ============================================
// 会话模型
// ============================================
model Session {
  id                String   @id @default(uuid())
  ownerId           String   @db.Uuid
  name              String   @db.VarChar(200)
  description       String?  @db.Text
  passwordHash      String?  @db.VarChar(255)
  controlMode       String   @default("request") @db.VarChar(20) // 'exclusive', 'request', 'free'
  autoAcceptControl Boolean  @default(false)
  allowChat         Boolean  @default(true)
  allowClipboard    Boolean  @default(false)
  status            String   @default("created") @db.VarChar(20) // 'created', 'active', 'paused', 'ended'
  expiresAt         DateTime?
  createdAt         DateTime @default(now())
  updatedAt         DateTime @updatedAt

  // 关联关系
  owner       User                @relation("SessionOwner", fields: [ownerId], references: [id], onDelete: Cascade)
  participants SessionParticipant[]
  events      SessionEvent[]

  // 索引
  @@index([ownerId])
  @@index([status])
  @@index([createdAt])
  @@index([expiresAt])
  @@map("sessions")
}

// ============================================
// 会话参与者模型
// ============================================
model SessionParticipant {
  id           String   @id @default(uuid())
  sessionId    String   @db.Uuid
  userId       String   @db.Uuid
  role         String   @default("guest") @db.VarChar(20) // 'owner', 'guest'
  controlState String   @default("none") @db.VarChar(20) // 'owner_hold', 'guest_hold', 'queued', 'none'
  joinedAt     DateTime @default(now())
  lastActiveAt DateTime @default(now())

  // 关联关系
  session Session @relation(fields: [sessionId], references: [id], onDelete: Cascade)
  user    User    @relation(fields: [userId], references: [id], onDelete: Cascade)

  // 索引
  @@index([sessionId])
  @@index([userId])
  @@unique([sessionId, userId])
  @@map("session_participants")
}

// ============================================
// 会话事件模型
// ============================================
model SessionEvent {
  id        String   @id @default(uuid())
  sessionId String   @db.Uuid
  userId    String?  @db.Uuid
  eventType String   @db.VarChar(50) // 'join', 'leave', 'control_request', 'control_grant', 'control_revoke', 'send', 'clear', 'mark'
  payload   Json?
  createdAt DateTime @default(now())

  // 关联关系
  session Session  @relation(fields: [sessionId], references: [id], onDelete: Cascade)
  user    User?   @relation(fields: [userId], references: [id], onDelete: SetNull)

  // 索引
  @@index([sessionId])
  @@index([sessionId, createdAt])
  @@index([eventType])
  @@map("session_events")
}
```

## 3. 数据库索引策略

### 3.1 主键索引

所有表使用 UUID 作为主键，自动创建主键索引。

### 3.2 唯一索引

| 表名 | 字段 | 说明 |
|------|------|------|
| `users` | `email` | 邮箱唯一性 |
| `serial_profiles` | `(userId, name)` | 用户配置名称唯一 |
| `session_participants` | `(sessionId, userId)` | 用户在同一会话中唯一 |

### 3.3 普通索引

#### 用户相关
- `users.email` - 登录查询
- `users.authProvider` - OAuth 查询
- `users.createdAt` - 用户统计

#### 配置相关
- `serial_profiles.userId` - 查询用户配置列表
- `serial_profiles(userId, isDefault)` - 查询默认配置
- `serial_profiles.deviceId` - 设备匹配

#### 会话相关
- `sessions.ownerId` - 查询用户发起的会话
- `sessions.status` - 查询活跃会话
- `sessions.createdAt` - 会话排序
- `sessions.expiresAt` - 过期会话清理
- `session_participants.sessionId` - 查询会话参与者
- `session_participants.userId` - 查询用户参与的会话
- `session_events(sessionId, createdAt)` - 查询会话事件时间线

#### 历史记录相关
- `send_histories(userId, createdAt)` - 查询用户历史（按时间排序）

### 3.4 索引优化建议

1. **定期分析**：每月运行 `ANALYZE` 更新统计信息
2. **监控慢查询**：启用 PostgreSQL 慢查询日志
3. **分区表**：`send_histories` 和 `session_events` 考虑按月分区（数据量大时）
4. **部分索引**：`serial_profiles` 的 `(userId, isDefault)` 可创建部分索引 `WHERE isDefault = true`

## 4. 数据验证规则

### 4.1 用户模型（User）

| 字段 | 验证规则 | 错误提示 |
|------|---------|---------|
| `email` | - 格式验证：正则 `/^[^\s@]+@[^\s@]+\.[^\s@]+$/`<br>- 长度：3-255 字符<br>- 唯一性：数据库约束 | "邮箱格式不正确"<br>"邮箱已被注册" |
| `passwordHash` | - 邮箱登录必填<br>- OAuth 登录可为空<br>- 哈希长度：60 字符（bcrypt） | "密码不能为空" |
| `nickname` | - 长度：1-100 字符<br>- 允许中文、英文、数字、下划线 | "昵称长度必须在 1-100 字符之间" |
| `avatar` | - URL 格式验证<br>- 长度：≤500 字符 | "头像 URL 格式不正确" |
| `authProvider` | - 枚举值：'email', 'google', 'github'<br>- 默认：'email' | "认证方式不支持" |

### 4.2 串口配置模型（SerialProfile）

| 字段 | 验证规则 | 错误提示 |
|------|---------|---------|
| `name` | - 长度：1-100 字符<br>- 同一用户下名称唯一 | "配置名称不能为空"<br>"配置名称已存在" |
| `baudRate` | - 范围：300-4000000<br>- 常用值：9600, 19200, 38400, 57600, 115200, 230400, 460800, 921600 | "波特率必须在 300-4000000 之间" |
| `dataBits` | - 枚举值：5, 6, 7, 8<br>- 默认：8 | "数据位必须是 5, 6, 7 或 8" |
| `parity` | - 枚举值：'none', 'even', 'odd', 'mark', 'space'<br>- 默认：'none' | "校验位不支持" |
| `stopBits` | - 枚举值：1, 1.5, 2<br>- 默认：1.0 | "停止位必须是 1, 1.5 或 2" |
| `flowControl` | - 枚举值：'none', 'hardware', 'software'<br>- 默认：'none' | "流控制方式不支持" |
| `lineEndingSend` | - 枚举值：'none', 'cr', 'lf', 'crlf' | "行结束符不支持" |
| `lineEndingParse` | - 枚举值：'none', 'cr', 'lf', 'crlf', 'auto' | "行结束符解析方式不支持" |
| `encoding` | - 枚举值：'utf-8', 'ascii', 'hex'<br>- 默认：'utf-8' | "编码格式不支持" |
| `bufferSize` | - 范围：1024-10485760 (10MB)<br>- 默认：65536 | "缓冲区大小必须在 1KB-10MB 之间" |
| `isDefault` | - 布尔值<br>- 同一用户最多一个默认配置 | "用户已存在默认配置" |

### 4.3 宏命令模型（Macro）

| 字段 | 验证规则 | 错误提示 |
|------|---------|---------|
| `name` | - 长度：1-100 字符 | "宏名称不能为空" |
| `category` | - 长度：1-50 字符<br>- 默认：'default' | "分类不能为空" |
| `contentTemplate` | - 长度：≤10000 字符<br>- 必须包含可解析的变量语法 `{variableName}` | "宏内容不能为空" |
| `variablesSchema` | - JSON 数组格式<br>- 每个变量包含：name, type, default, description | "变量定义格式不正确" |

**变量 Schema 示例**：
```json
[
  {
    "name": "deviceId",
    "type": "number",
    "default": 0,
    "description": "设备 ID"
  },
  {
    "name": "command",
    "type": "string",
    "default": "STATUS",
    "description": "命令名称"
  }
]
```

### 4.4 会话模型（Session）

| 字段 | 验证规则 | 错误提示 |
|------|---------|---------|
| `name` | - 长度：1-200 字符 | "会话名称不能为空" |
| `controlMode` | - 枚举值：'exclusive', 'request', 'free'<br>- 默认：'request' | "控制权模式不支持" |
| `status` | - 枚举值：'created', 'active', 'paused', 'ended'<br>- 默认：'created' | "会话状态无效" |
| `expiresAt` | - 必须晚于当前时间<br>- 最长 30 天 | "过期时间无效" |

### 4.5 数据库约束

#### 外键约束
- 所有外键使用 `ON DELETE CASCADE`，删除用户时自动删除关联数据
- `SessionEvent.userId` 使用 `ON DELETE SET NULL`，允许系统事件无用户

#### 检查约束（可选，应用层也可实现）
```sql
-- 波特率范围检查
ALTER TABLE serial_profiles ADD CONSTRAINT check_baud_rate 
  CHECK (baud_rate >= 300 AND baud_rate <= 4000000);

-- 数据位检查
ALTER TABLE serial_profiles ADD CONSTRAINT check_data_bits 
  CHECK (data_bits IN (5, 6, 7, 8));

-- 停止位检查
ALTER TABLE serial_profiles ADD CONSTRAINT check_stop_bits 
  CHECK (stop_bits IN (1, 1.5, 2));
```

## 7. 数据关系说明

### 7.1 一对多关系

- **User → SerialProfile**：一个用户可以有多个串口配置
- **User → Macro**：一个用户可以有多个宏命令
- **User → SendHistory**：一个用户可以有多个发送历史记录
- **Session → SessionParticipant**：一个会话可以有多个参与者
- **Session → SessionEvent**：一个会话可以有多个事件

### 7.2 一对一关系

- **User → UserPreference**：一个用户只有一份偏好配置

### 7.3 级联删除规则

- 删除用户时，自动删除：所有配置、宏、历史、偏好、会话（作为 owner）
- 删除会话时，自动删除：所有参与者、所有事件
- 删除会话事件时，如果用户已删除，userId 设置为 NULL

---

## 8. 性能优化建议

### 8.1 查询优化

1. **批量查询**：使用 `include` 或 `select` 减少 N+1 查询
2. **分页查询**：所有列表接口必须支持分页（cursor 或 offset）
3. **字段选择**：只查询需要的字段，避免 `SELECT *`

### 8.2 数据归档

- `send_histories`：超过 200 条的旧记录归档到归档表
- `session_events`：超过 90 天的会话事件归档

### 8.3 缓存策略

- 用户默认配置：Redis 缓存（TTL: 1 小时）
- 用户偏好：Redis 缓存（TTL: 1 小时）
- 活跃会话列表：Redis 缓存（TTL: 5 分钟）
